async page => {
 await page.setViewportSize({width:1440,height:960});
 await page.goto('http://localhost:3000/inventory?q=burst&page=1');
 await page.waitForLoadState('networkidle');
 await page.getByRole('link',{name:/Detail barang/}).first().click();
 const panel=page.getByRole('dialog',{name:/^Unit fisik/});await panel.waitFor();
 const selected=page.url();
 await page.getByRole('button',{name:'Tambah jenis barang',exact:true}).click();
 const create=page.getByRole('dialog',{name:'Tambah jenis barang',exact:true});await create.waitFor();
 await page.keyboard.press('Escape');await create.waitFor({state:'hidden'});
 if(!(await panel.isVisible())||page.url()!==selected)throw new Error('Create Escape changed existing unit selection');
 if(await page.evaluate(()=>document.body.style.overflow==='hidden'))throw new Error('Body remains locked');
 await panel.getByRole('button',{name:'Tutup daftar unit'}).click();await panel.waitFor({state:'hidden'});
 await page.waitForURL(u=>!u.searchParams.has('item'));
 if(new URL(page.url()).searchParams.get('q')!=='burst')throw new Error('Search lost');
 return {nestedEscape:'only form closed; unit selection preserved',search:'burst preserved'};
}
