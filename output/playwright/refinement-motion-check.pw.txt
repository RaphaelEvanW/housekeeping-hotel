async page => {
  const checks=[];
  const assert=(v,label)=>{if(!v)throw new Error(label);checks.push(label);};
  await page.setViewportSize({width:1280,height:900});
  await page.goto('http://localhost:3000/inventory');
  await page.waitForLoadState('networkidle');
  const cards=page.getByRole('list',{name:'Katalog jenis barang'});
  const before=await cards.evaluate(e=>{const rows=[...e.children].map(c=>c.getBoundingClientRect());return{width:rows[0].width,columns:rows.filter(r=>r.top===rows[0].top).length};});
  const frames=page.evaluate(()=>new Promise(resolve=>{const d=[];let previous=performance.now();function tick(now){d.push(now-previous);previous=now;if(d.length<40)requestAnimationFrame(tick);else resolve({frames:d.length,over50ms:d.filter(x=>x>50).length,maximumMs:Math.round(Math.max(...d))});}requestAnimationFrame(tick);}));
  await page.getByRole('button',{name:'Ciutkan sidebar'}).click();
  await page.waitForFunction(()=>Math.round(document.querySelector('aside').getBoundingClientRect().width)===72);
  const after=await cards.evaluate(e=>{const rows=[...e.children].map(c=>c.getBoundingClientRect());return{width:rows[0].width,columns:rows.filter(r=>r.top===rows[0].top).length};});
  assert(before.width===after.width,'card width unchanged when sidebar collapses');
  assert(before.columns===3&&after.columns===4,'collapsed sidebar adds fourth column at 1280px');
  const frameResult=await frames;
  await page.getByRole('button',{name:'Perluas sidebar'}).click();
  await page.waitForFunction(()=>Math.round(document.querySelector('aside').getBoundingClientRect().width)===288);
  const styles=await cards.locator('h3').first().evaluate(e=>({font:getComputedStyle(e).fontFamily,weight:getComputedStyle(e).fontWeight,spacing:getComputedStyle(e).letterSpacing}));
  assert(styles.font.includes('Inter')&&styles.weight==='500'&&styles.spacing==='normal','Inter medium with normal tracking on item names');
  assert(await page.getByRole('complementary',{name:'Sidebar desktop'}).getByText('Selamat datang',{exact:true}).isVisible(),'welcome copy appears in sidebar');
  const glass=await page.getByRole('complementary',{name:'Sidebar desktop'}).locator('.sidebar-glass').evaluateAll(es=>es.map(e=>getComputedStyle(e).backdropFilter));
  assert(glass.length===3&&glass.every(s=>s==='blur(10px)'),'three localized glass surfaces');
  const nav=page.getByRole('complementary',{name:'Sidebar desktop'}).getByRole('navigation',{name:'Navigasi utama'});
  const indicator=nav.locator('[aria-hidden="true"].sidebar-glass');
  const motionDuration=await indicator.evaluate(e=>getComputedStyle(e).transitionDuration);
  assert(motionDuration==='0.38s','shared active indicator has 380ms slide');
  await nav.getByRole('link',{name:'Dashboard Operasional'}).click();
  await page.waitForURL('**/dashboard');
  await page.waitForFunction(()=>document.querySelector('aside nav [aria-hidden="true"].sidebar-glass').style.transform==='translateY(0px)');
  await page.goto('http://localhost:3000/inventory');
  await page.waitForLoadState('networkidle');
  await cards.getByRole('link',{name:/Lihat unit/}).first().click();
  const panel=page.getByRole('dialog',{name:/^Unit fisik/});
  await panel.waitFor();
  const panelDuration=await panel.evaluate(e=>e.getAnimations().map(a=>a.effect.getTiming().duration));
  await panel.getByRole('button',{name:'Tutup daftar unit'}).click();
  await panel.waitFor({state:'hidden'});
  assert(await page.evaluate(()=>document.body.style.overflow!=='hidden'),'body scroll restored as panel closes');
  await page.waitForURL(u=>!u.searchParams.has('item'));
  await page.setViewportSize({width:390,height:900});
  await page.getByRole('button',{name:'Buka sidebar'}).click();
  const menu=page.getByRole('dialog',{name:'Mövenpick Hotel',exact:true});
  await menu.waitFor();
  await menu.getByRole('button',{name:'Tutup sidebar'}).click();
  await menu.waitFor({state:'hidden'});
  await page.emulateMedia({reducedMotion:'reduce'});
  await page.getByRole('button',{name:'Buka sidebar'}).click();
  await menu.waitFor();
  assert((await menu.evaluate(e=>e.getAnimations().length))===0,'reduced motion skips panel movement');
  await page.keyboard.press('Escape');
  await menu.waitFor({state:'hidden'});
  await page.emulateMedia({reducedMotion:'no-preference'});
  return {checks,before,after,frameResult,styles,panelDuration};
}
