async page => {
  const checks = [];
  const assert = (ok, label) => { if (!ok) throw new Error(label); checks.push(label); };
  await page.setViewportSize({width:1280,height:900});
  await page.goto('http://localhost:3000/inventory');
  const catalog = page.getByRole('list', {name:'Katalog jenis barang'});
  await catalog.waitFor();
  const next = page.getByRole('navigation', {name:'Halaman jenis barang'}).getByRole('link', {name:'Berikutnya'});
  await next.click();
  await page.waitForURL(/page=2/);
  assert(new URL(page.url()).searchParams.get('page') === '2', 'item pagination next');
  await page.getByRole('navigation', {name:'Halaman jenis barang'}).getByRole('link', {name:'Sebelumnya'}).click();
  await page.waitForURL(/page=1/);
  assert(new URL(page.url()).searchParams.get('page') === '1', 'item pagination previous');
  await page.getByRole('button', {name:'Tabel',exact:true}).click();
  assert(await page.getByRole('table').isVisible(), 'table view visible');
  assert(!(await catalog.isVisible()), 'card view hidden in table mode');
  await page.getByRole('button', {name:'Kartu',exact:true}).click();
  assert(await catalog.isVisible(), 'card view restored');
  const names = await catalog.locator('h3').allTextContents();
  const shortName = names.find(n=>n.length > 0 && n.length < 100);
  assert(!!shortName, 'existing short item available for search');
  await page.getByRole('textbox', {name:'Cari nama barang'}).fill(shortName);
  await page.getByRole('button', {name:'Cari',exact:true}).click();
  await page.waitForURL(url=>url.searchParams.has('q'));
  await catalog.waitFor();
  assert((await catalog.locator('h3').allTextContents()).every(n=>n.toLowerCase().includes(shortName.toLowerCase())), 'search results match existing item');
  await catalog.getByRole('link', {name:/Lihat unit/}).first().click();
  await page.waitForURL(url=>url.searchParams.has('item'));
  await page.getByRole('heading', {name:/^Unit /}).waitFor();
  assert(!!new URL(page.url()).searchParams.get('q'), 'selection preserves search');
  await page.getByRole('link', {name:'Ke daftar unit',exact:true}).click();
  await page.screenshot({path:'output/playwright/inventory-selected-1280.png'});
  checks.push('selected item opens unit section');
  const unitFacts = await page.locator('[aria-labelledby="units-heading"]').innerText();
  checks.push(unitFacts.includes('Belum ada unit fisik') ? 'selected item has no physical units; cursor not exercised' : 'existing physical unit facts rendered');
  await page.getByRole('link', {name:'Tutup daftar unit',exact:true}).click();
  await page.waitForURL(url=>!url.searchParams.has('item'));
  assert(!!new URL(page.url()).searchParams.get('q'), 'closing units preserves search');
  await page.getByRole('textbox', {name:'Cari nama barang'}).fill('zz-no-match-visual-review-20261022');
  await page.getByRole('button', {name:'Cari',exact:true}).click();
  await page.getByRole('heading', {name:'Barang tidak ditemukan',exact:true}).waitFor();
  checks.push('empty search state');
  await page.getByRole('link', {name:'Hapus pencarian',exact:true}).click();
  await catalog.waitFor();
  assert(!new URL(page.url()).searchParams.has('q'), 'clear search');
  await page.setViewportSize({width:390,height:900});
  await page.getByText('Menu navigasi',{exact:true}).click();
  assert(await page.getByRole('link',{name:'Dashboard Operasional',exact:true}).isVisible(), 'mobile navigation opens');
  await page.screenshot({path:'output/playwright/navigation-mobile.png'});
  await page.getByRole('link',{name:'Dashboard Operasional',exact:true}).click();
  await page.waitForURL('**/dashboard');
  checks.push('mobile navigation routes to dashboard');
  return checks;
}
