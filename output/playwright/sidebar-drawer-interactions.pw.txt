async page => {
  const checks=[];
  const assert=(v,name)=>{if(!v)throw new Error(name);checks.push(name);};
  await page.setViewportSize({width:1280,height:900});
  await page.goto('http://localhost:3000/inventory?page=2');
  await page.waitForLoadState('networkidle');
  const catalog=page.getByRole('list',{name:'Katalog jenis barang'});
  const name=await catalog.locator('h3').first().textContent();
  await page.getByRole('textbox',{name:'Cari nama barang'}).fill(name.slice(0,20));
  await page.getByRole('button',{name:'Cari',exact:true}).click();
  await page.waitForURL(u=>u.searchParams.has('q'));
  await catalog.waitFor();
  const expectedQuery=new URL(page.url()).searchParams.get('q').trim().toLowerCase().replace(/\s+/g,' ');
  await catalog.getByRole('link',{name:/Lihat unit/}).first().click();
  const drawer=page.getByRole('dialog',{name:/^Unit /});
  await drawer.waitFor();
  await drawer.getByRole('button',{name:'Tutup daftar unit'}).focus();
  for(let i=0;i<8;i++)await page.keyboard.press('Tab');
  assert(await drawer.evaluate(e=>e.contains(document.activeElement)),'modal traps tab focus');
  await drawer.getByRole('button',{name:'Tutup daftar unit'}).click();
  await drawer.waitFor({state:'hidden'});
  assert(new URL(page.url()).searchParams.get('q')===expectedQuery,'close preserves search');
  assert(await page.evaluate(()=>document.activeElement?.getAttribute('aria-label')?.startsWith('Lihat unit')),'close restores item action focus');
  await catalog.getByRole('link',{name:/Lihat unit/}).first().click();
  await drawer.waitFor();
  await page.mouse.click(10,300);
  await drawer.waitFor({state:'hidden'});
  checks.push('backdrop closes unit panel');
  await catalog.getByRole('link',{name:/Lihat unit/}).first().click();
  await drawer.waitFor();
  await page.setViewportSize({width:1440,height:900});
  await page.waitForFunction(()=>!document.querySelector('dialog[aria-labelledby="units-heading"]').matches(':modal'));
  assert(await page.evaluate(()=>document.body.style.overflow!=='hidden'),'resize modal to desktop releases scroll');
  await page.setViewportSize({width:768,height:900});
  await page.waitForFunction(()=>document.querySelector('dialog[aria-labelledby="units-heading"]').matches(':modal'));
  assert(await page.evaluate(()=>document.body.style.overflow==='hidden'),'resize desktop to tablet locks background');
  await page.keyboard.press('Escape');
  await drawer.waitFor({state:'hidden'});
  await page.getByRole('button',{name:'Buka sidebar'}).click();
  const menu=page.getByRole('dialog',{name:'Mövenpick Hotel',exact:true});
  await menu.waitFor();
  await page.mouse.click(740,300);
  await menu.waitFor({state:'hidden'});
  checks.push('backdrop closes mobile sidebar');
  await page.getByRole('button',{name:'Buka sidebar'}).click();
  await menu.waitFor();
  await page.setViewportSize({width:1440,height:900});
  await menu.waitFor({state:'hidden'});
  assert(await page.evaluate(()=>document.body.style.overflow!=='hidden'),'resize to desktop closes sidebar and releases scroll');
  return checks;
}
